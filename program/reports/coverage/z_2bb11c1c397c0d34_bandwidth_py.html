1"""Report-only content bandwidth of one audio file: where its audio really stops. 

2 

3A lossy encoder removes everything above its low-pass frequency, and the gap 

4survives decoding. A WAV or FLAC made from an MP3 still stops where the MP3 

5did; an MP3 re-encoded at a higher bitrate still stops where the lower bitrate 

6did; audio resampled up to 96 kHz has nothing above the original Nyquist 

7frequency. The long-term spectrum of the decoded audio is compared with what 

8the container and bitrate claim. 

9 

10This is a heuristic, never a score. A band limit says where the audio stops, 

11not why: AI generators produce band-limited audio natively (all 25 Lumina 

12benchmark WAVs stop between 17.4 and 19.9 kHz), mastering can low-pass a recording, 

13quiet material can hide a limit in its noise, and none of it shows who made a 

14file. The messages therefore name the alternatives instead of a verdict. 

15"""

39def spectrum_profile(blocks, total_frames, rate, parameters): 

40 """90th-percentile power spectrum (dB) of ``spectrum_frames`` frames spread evenly over the whole file. 

41 

42 Frames are sampled across the full duration (not a prefix), so the count 

43 bounds cost without limiting which part of the file is described. 

44 """ 

45 size, limit = parameters["frame_size"], parameters["spectrum_frames"] 

46 if total_frames < size: 

47 raise AnalysisUnavailable("insufficient_signal", "Too little audio to measure its bandwidth.") 

48 stride = max(size, total_frames // limit) 

49 window = np.hanning(size) 

50 rows, carry, position, next_start = [], np.zeros(0, dtype=np.float32), 0, 0 

51 for block in blocks(0, total_frames): 

52 carry = np.concatenate((carry, block.mean(axis=1))) 

53 while next_start + size <= position + len(carry) and len(rows) < limit: 

54 offset = next_start - position 

55 frame = carry[offset:offset + size] 

56 if np.sqrt(np.mean(frame.astype(np.float64) ** 2)) >= 1e-4: 56 ↛ 58line 56 didn't jump to line 58 because the condition on line 56 was always true

57 rows.append(np.abs(np.fft.rfft(frame * window)) ** 2) 

58 next_start += stride 

59 keep = max(0, next_start - position) 

60 position += min(keep, len(carry)) 

61 carry = carry[min(keep, len(carry)):] 

62 if len(rows) < 4: 62 ↛ 63line 62 didn't jump to line 63 because the condition on line 62 was never true

63 raise AnalysisUnavailable("insufficient_signal", "Too few non-silent frames to measure bandwidth.") 

64 level = np.percentile(np.array(rows), parameters["level_percentile"], axis=0) 

65 freqs = np.fft.rfftfreq(size, 1 / rate) 

66 db = 10 * np.log10(level + 1e-20) 

67 reference = float(np.median(db[(freqs >= 1000) & (freqs < 4000)])) 

68 return freqs, db - reference, len(rows)

75def _transition(freqs, rel, cutoff): 

76 """Where, within 1 kHz of a candidate, the level first falls halfway from below to above.""" 

77 middle = (cutoff["level_below_db"] + cutoff["level_above_db"]) / 2 

78 window = (freqs >= cutoff["frequency_hz"] - 1000) & (freqs <= cutoff["frequency_hz"] + 1000) 

79 span = freqs[window] 

80 # Smooth over about 100 Hz so a single noisy bin cannot set the edge. 

81 width = max(1, int(round(100 / (freqs[1] - freqs[0])))) 

82 level = np.convolve(rel[window], np.ones(width) / width, mode="same") 

83 below_middle = np.nonzero(level < middle)[0] 

84 return float(span[below_middle[0]]) if len(below_middle) else cutoff["frequency_hz"]

87def find_cutoffs(freqs, rel, rate, parameters): 

88 """The strongest sharp cliff and the strongest step into a flat noise floor, if any.""" 

89 nyquist = rate / 2 

90 low = float(np.ceil(max(4000.0, 0.35 * nyquist) / 100.0) * 100.0) 

91 cliff = step = None 

92 for fc in np.arange(low, nyquist - 600, 100.0): 

93 below = _median(rel[(freqs >= fc - 1000) & (freqs < fc - 150)]) 

94 above = _median(rel[(freqs >= fc + 150) & (freqs < min(fc + 2000, nyquist - 100))]) 

95 rest = rel[(freqs >= fc + 300) & (freqs < nyquist - 100)] 

96 if not len(rest) or np.isnan(above): 96 ↛ 97line 96 didn't jump to line 97 because the condition on line 96 was never true

97 continue 

98 drop = below - above 

99 if (drop >= parameters["cliff_drop_db"] and float(np.percentile(rest, 95)) <= below - STRONG_FLOOR_MARGIN_DB 

100 and (cliff is None or drop > cliff["drop_db"])): 

101 cliff = {"kind": "sharp_cliff", "frequency_hz": float(fc), "drop_db": round(drop, 1), 

102 "level_below_db": round(below, 1), "level_above_db": round(above, 1)} 

103 top = parameters["step_search_top_fraction"] * nyquist 

104 for fc in np.arange(low, top - 1000, 100.0): 

105 below = _median(rel[(freqs >= fc - 1500) & (freqs < fc - 500)]) 

106 bands = [_median(rel[(freqs >= f) & (freqs < f + 500)]) for f in np.arange(fc + 500, top, 500.0)] 

107 if len(bands) < 2: 107 ↛ 108line 107 didn't jump to line 108 because the condition on line 107 was never true

108 continue 

109 above, spread = float(np.median(bands)), float(max(bands) - min(bands)) 

110 drop = below - above 

111 if (drop >= parameters["step_drop_db"] and spread <= parameters["flat_floor_spread_db"] 

112 and (step is None or drop > step["drop_db"])): 

113 step = {"kind": "step_into_flat_floor", "frequency_hz": float(fc), "drop_db": round(drop, 1), 

114 "level_below_db": round(below, 1), "level_above_db": round(above, 1), 

115 "floor_spread_db": round(spread, 1)} 

116 found = [item for item in (cliff, step) if item] 

117 for item in found: 

118 item["frequency_hz"] = round(_transition(freqs, rel, item), 1) 

119 return found

129def assess(edge, rate, container, bitrate_kbps=None): 

130 """Compare the content edge with what the container and bitrate claim.""" 

131 nyquist = rate / 2 

132 if container == "MP3": 

133 expected = typical_lowpass(bitrate_kbps or 0) if rate >= 32000 else None 

134 if expected is None: 

135 return {"status": "not_assessed", "reason": "No typical low-pass frequency is known for this MP3's " 

136 "sample rate or bitrate."} 

137 if edge is not None and edge < expected - 1500: 

138 return {"status": "content_band_lower_than_bitrate_suggests", "typical_lowpass_hz": expected, 

139 "reason": f"The audio stops at {edge / 1000:.1f} kHz, but a {bitrate_kbps} kbps MP3 usually " 

140 f"reaches about {expected / 1000:.1f} kHz: it was probably re-encoded from a lower " 

141 "bitrate, or its source was already band-limited."} 

142 return {"status": "consistent_with_bitrate", "typical_lowpass_hz": expected, 

143 "reason": "The content band matches what this bitrate usually keeps."} 

144 if edge is None: 

145 return {"status": "no_band_limit_found", 

146 "reason": "No sharp band limit below the Nyquist frequency was found."} 

147 base = next((value for value in STANDARD_RATES 

148 # Content ending just below a lower standard rate's Nyquist frequency. 

149 if value < rate and 0.9 * value / 2 <= edge <= 1.01 * value / 2), None) 

150 if 11000 <= edge <= 20500 and edge < 0.9 * nyquist: 

151 # Checked first: lossy low-passes are far more common in music than 

152 # material resampled up from 22.05-32 kHz, whose Nyquist falls here too. 

153 also = f", or resampled up from {base} Hz audio" if base else "" 

154 return {"status": "band_limit_like_lossy_encoding", 

155 "reason": f"This {container} file is lossless, but its audio stops at {edge / 1000:.1f} kHz with a " 

156 "sharp limit like a lossy encoder's low-pass. It may have been made from an MP3 or similar " 

157 f"file{also}, or generated or filtered to this bandwidth (AI generators often are)."} 

158 if base: 

159 return {"status": "consistent_with_upsampling", "likely_source_rate_hz": base, 

160 "reason": f"The file is sampled at {rate} Hz, but its audio stops at {edge / 1000:.2f} kHz, the " 

161 f"Nyquist frequency of {base} Hz audio: it was probably resampled up from {base} Hz, " 

162 "or its source never had content above that band."} 

163 return {"status": "band_limited", 

164 "reason": f"The audio stops at {edge / 1000:.1f} kHz, below the Nyquist frequency of " 

165 f"{nyquist / 1000:.1f} kHz."}

168def analyse(task, parameters, context=None): 

169 path = task["source_path"] 

170 from music_target.audio_formats import detect 

171 from music_target.mp3_io import inspect_mp3 

172 container = detect(path).container 

173 bitrate = inspect_mp3(path).nominal_bitrate_kbps if container == "MP3" else None 

174 descriptor, blocks = _open(path) 

175 freqs, rel, frames_used = spectrum_profile(blocks, descriptor.frame_count, descriptor.sample_rate, parameters) 

176 cutoffs = find_cutoffs(freqs, rel, descriptor.sample_rate, parameters) 

177 edge = min((item["frequency_hz"] for item in cutoffs), default=None) 

178 assessment = assess(edge, descriptor.sample_rate, container, bitrate) 

179 value = result(task, "bandwidth", "bandwidth_observation", assessment["reason"]) 

180 value.update( 

181 analysis_complete=True, 

182 analysed_ranges={"whole_file": [0.0, descriptor.duration_seconds], 

183 "original_sample_rates_hz": [descriptor.sample_rate], 

184 "channel_policy": "arithmetic_mean"}, 

185 metrics={"container": container, "sample_rate_hz": descriptor.sample_rate, 

186 "nyquist_hz": descriptor.sample_rate / 2, "nominal_bitrate_kbps": bitrate, 

187 "content_edge_hz": edge, "cutoffs": cutoffs, "frames_analysed": frames_used, 

188 "assessment": assessment}, 

189 limitations=list(LIMITATIONS)) 

190 return value
