20def _windowed_peak_search(power, parameters): 

21 # Upstream derives its peak distance from the size of the array it is 

22 # given, so one call over a whole file makes peaks sparser as the file 

23 # gets longer and a short excerpt stops sharing peaks with a long source. 

24 # Fixed-duration, zero-padded windows keep that distance constant in 

25 # seconds for every file; padding is all-zero and yields no peaks. 

26 from spectromap.spectromap import peak_search 

27 width = window_frames(parameters) 

28 mask = np.zeros(power.shape, dtype=bool) 

29 for start in range(0, power.shape[1], width): 

30 block = power[:, start:start + width] 

31 padded = np.zeros((power.shape[0], width), dtype=power.dtype) 

32 padded[:, :block.shape[1]] = block 

33 found, _ = peak_search(padded, fraction=parameters["fraction"], condition=parameters["condition"]) 

34 mask[:, start:start + block.shape[1]] = found[:, :block.shape[1]] 

35 return mask
