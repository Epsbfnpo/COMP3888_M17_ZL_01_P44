265def _fir_lowpass(rate, cutoff_hz, transition_scale=1.0):

268 return np.ones(1, dtype=np.float32) 

269 ratio = max(1.0, rate / (2.0 * cutoff_hz)) 

270 half = max(8, math.ceil(PROXY_FILTER_HALF_WIDTH * ratio * transition_scale)) 

271 positions = np.arange(-half, half + 1, dtype=np.float64) 

272 normalized_cutoff = cutoff_hz / rate 

273 taps = (2.0 * normalized_cutoff * 

274 np.sinc(2.0 * normalized_cutoff * positions) *

414def _source_identifiability(gram): 

415 """Return a scale-free pivot ratio for distinguishing source contributions.""" 

416 gram = np.asarray(gram, dtype=np.float64) 

417 size = gram.shape[0] if gram.ndim == 2 else 0 

418 if not size: 

419 return 0.0 

420 diagonal = np.maximum(np.diag(gram), 0.0) 

421 denominator = np.sqrt(np.outer(diagonal, diagonal)) 

422 normalized = np.divide(gram, denominator, out=np.zeros_like(gram), 

423 where=denominator > 1e-12) 

424 singular = np.linalg.svd(normalized, compute_uv=False) 

425 if singular.size == 0 or singular[0] < 1e-12: 

426 return 0.0 

427 return float(singular[-1] / singular[0])

487def _update_partition(statistics, actual, predicted): 

488 actual = np.asarray(actual, dtype=np.float64) 

489 predicted = np.asarray(predicted, dtype=np.float64) 

490 statistics["count"] += actual.size 

491 statistics["actual_sum"] += float(np.sum(actual)) 

492 statistics["predicted_sum"] += float(np.sum(predicted)) 

493 statistics["actual_square"] += float(np.sum(actual * actual, dtype=np.float64)) 

494 statistics["predicted_square"] += float(np.sum(predicted * predicted, dtype=np.float64)) 

495 statistics["cross"] += float(np.sum(actual * predicted, dtype=np.float64)) 

496 difference = actual - predicted 

497 statistics["residual_square"] += float( 

498 np.sum(difference * difference, dtype=np.float64))

501def _partition_result(statistics): 

502 count = statistics["count"] 

503 energy = statistics["actual_square"] 

504 if count < 1 or energy < 1e-12: 

505 return None 

506 actual_variance = energy - statistics["actual_sum"] ** 2 / count 

507 predicted_variance = (statistics["predicted_square"] - 

508 statistics["predicted_sum"] ** 2 / count) 

509 covariance = (statistics["cross"] - 

510 statistics["actual_sum"] * statistics["predicted_sum"] / count) 

511 denominator = math.sqrt(max(actual_variance, 0.0) * 

512 max(predicted_variance, 0.0)) 

513 return (math.sqrt(statistics["residual_square"] / energy), 

514 covariance / denominator if denominator > 1e-12 else 0.0)

530 declared_gains = [] 

531 keys = list(group_keys) if group_keys is not None else list(range(len(edges))) 

532 unique_keys = list(dict.fromkeys(keys)) 

533 key_positions = {key: position for position, key in enumerate(unique_keys)} 

534 for position, edge in enumerate(edges): 

535 attrs = edge.get("attributes") or {} 

536 source_path, source_meta = _bound_audio_path(chain, edge["hash"]) 

537 if (source_meta["sample_rate_hz"] != rate or 

538 source_meta["channels"] != channels): 

539 raise ValueError("source and target sample rates/channels must match") 

540 source_start = float(attrs["source_start_seconds"]) 

541 source_end = float(attrs["source_end_seconds"]) 

542 target_start = float(attrs["target_start_seconds"]) 

543 target_end = float(attrs["target_end_seconds"]) 

544 gain = float(attrs["gain"]) 

545 fade_in = float(attrs.get("fade_in_seconds") or 0.0) 

546 fade_out = float(attrs.get("fade_out_seconds") or 0.0) 

547 if not all(math.isfinite(value) for value in ( 

548 source_start, source_end, target_start, target_end, 

549 gain, fade_in, fade_out)): 

550 raise ValueError("declared linear parameters must be finite") 

551 if (min(source_start, target_start, fade_in, fade_out) < 0 or 

552 source_end <= source_start or target_end <= target_start): 

553 raise ValueError("declared source/target ranges must be positive and ordered") 

554 if abs((source_end - source_start) - (target_end - target_start)) > max(0.002, 2 / rate): 

555 raise ValueError("declared ranges imply time stretching") 

556 if source_end > source_meta["duration_seconds"] + 2 / rate: 

557 raise ValueError("declared source range exceeds the submitted audio") 

558 if target_end > target_meta["duration_seconds"] + 2 / rate: 

559 raise ValueError("declared target range exceeds the submitted audio") 

560 if fade_in + fade_out > target_end - target_start + 2 / rate: 

561 raise ValueError("declared fades must fit within the target range") 

562 offsets.append(target_start - source_start) 

563 declared_gains.append(gain) 

564 prepared.append({ 

565 "path": source_path, 

566 "source_start": source_start, 

567 "source_end": source_end, 

568 "target_start": target_start, 

569 "target_end": target_end, 

570 "gain": gain, 

571 "fade_in": fade_in, 

572 "fade_out": fade_out, 

573 "group": key_positions[keys[position]], 

574 "metadata": source_meta, 

575 "source_start_frame": round(source_start * rate), 

576 "source_end_frame": round(source_end * rate), 

577 "target_start_frame": round(target_start * rate), 

578 "target_end_frame": round(target_end * rate), 

579 }) 

580 

581 if target_meta["total_frames"] < 32 or not unique_keys: 

582 return None 

583 _consume_work(chain, target_meta["duration_seconds"] * channels * 

584 (1.0 + len(prepared))) 

585 holdout_frames = max(1, round(rate * HOLDOUT_BLOCK_SECONDS)) 

586 desired_chunk = max(holdout_frames, 

587 round(rate * PCM_STREAM_CHUNK_SECONDS / holdout_frames) * 

588 holdout_frames) 

589 bytes_per_frame = channels * np.dtype(np.float32).itemsize 

590 memory_chunk = max(holdout_frames, 

591 (MAX_ACTIVE_PCM_BYTES // 

592 max(bytes_per_frame * (len(unique_keys) + 3), 1) // 

593 holdout_frames) * holdout_frames) 

594 chunk_frames = min(desired_chunk, memory_chunk) 

595 if chunk_frames < holdout_frames: 

596 raise ResourceBudgetExceeded("one exact-analysis holdout block exceeds the memory limit") 

597 

598 empty_stat = lambda: { 

599 "count": 0, "actual_sum": 0.0, "predicted_sum": 0.0, 

600 "actual_square": 0.0, "predicted_square": 0.0, 

601 "cross": 0.0, "residual_square": 0.0, 

602 } 

603 fit_stat, validation_stat = empty_stat(), empty_stat() 

604 gram = np.zeros((len(unique_keys), len(unique_keys)), dtype=np.float64) 

605 covered_count = 0 

606 matched_frames = 0 

607 mismatch_regions = [] 

608 pending_region = None 

609 

610 def close_pending(): 

611 nonlocal pending_region 

612 if pending_region is not None: 

613 duration = pending_region["end_seconds"] - pending_region["start_seconds"] 

614 if duration + 1e-9 >= LOCAL_MISMATCH_MIN_SECONDS: 

615 mismatch_regions.append({**pending_region, 

616 "duration_seconds": duration}) 

617 pending_region = None 

618 

619 for block_start_frame in range(0, target_meta["total_frames"], chunk_frames): 

620 _check_deadline(chain) 

621 block_end_frame = min(target_meta["total_frames"], 

622 block_start_frame + chunk_frames) 

623 target_audio = _read_pcm_frame_range( 

624 target_path, block_start_frame, block_end_frame, target_meta) 

625 target_frames = target_audio["frames"][:block_end_frame - block_start_frame] 

626 frame_count = len(target_frames) 

627 grouped = np.zeros((len(unique_keys), frame_count, channels), dtype=np.float32) 

628 covered = np.zeros(frame_count, dtype=bool) 

629 for item in prepared: 

630 overlap_start_frame = max(block_start_frame, 

631 item["target_start_frame"]) 

632 overlap_end_frame = min(block_end_frame, 

633 item["target_end_frame"]) 

634 if overlap_end_frame <= overlap_start_frame: 

635 continue 

636 target_first = overlap_start_frame - block_start_frame 

637 target_last = overlap_end_frame - block_start_frame 

638 if target_last <= target_first: 

639 continue 

640 source_first_frame = (item["source_start_frame"] + 

641 overlap_start_frame - 

642 item["target_start_frame"]) 

643 source_last_frame = source_first_frame + (target_last - target_first) 

644 source_audio = _read_pcm_frame_range( 

645 item["path"], source_first_frame, source_last_frame, 

646 item["metadata"]) 

647 count = min(target_last - target_first, len(source_audio["frames"])) 

648 if count <= 0: 

649 continue 

650 global_frames = block_start_frame + target_first + np.arange(count) 

651 target_times = global_frames.astype(np.float64) / rate 

652 envelope = np.ones(count, dtype=np.float32) 

653 if item["fade_in"]: 

654 envelope = np.minimum( 

655 envelope, 

656 ((target_times - item["target_start"]) / 

657 item["fade_in"]).astype(np.float32)) 

658 if item["fade_out"]: 

659 envelope = np.minimum( 

660 envelope, 

661 ((item["target_end"] - target_times) / 

662 item["fade_out"]).astype(np.float32)) 

663 np.clip(envelope, 0.0, 1.0, out=envelope) 

664 grouped[item["group"], target_first:target_first + count] += ( 

665 source_audio["frames"][:count] * 

666 (np.float32(item["gain"]) * envelope[:, None])) 

667 covered[target_first:target_first + count] = True 

668 

669 predicted_frames = np.sum(grouped, axis=0, dtype=np.float64) 

670 target64 = target_frames.astype(np.float64, copy=False) 

671 global_frame_numbers = block_start_frame + np.arange(frame_count) 

672 fit_frame_mask = ((global_frame_numbers // holdout_frames) % 2) == 0 

673 for mask, statistics in ((fit_frame_mask, fit_stat), 

674 (~fit_frame_mask, validation_stat)): 

675 sample_mask = np.repeat(mask, channels) 

676 _update_partition(statistics, target64.reshape(-1)[sample_mask], 

677 predicted_frames.reshape(-1)[sample_mask]) 

678 fit_grouped = grouped[:, fit_frame_mask, :].reshape(len(unique_keys), -1) 

679 fit_grouped64 = fit_grouped.astype(np.float64) 

680 gram += np.einsum("in,jn->ij", fit_grouped64, fit_grouped64, 

681 optimize=True) 

682 covered_count += int(np.count_nonzero(covered)) 

683 

684 for local_start in range(0, frame_count, holdout_frames): 

685 local_end = min(frame_count, local_start + holdout_frames) 

686 covered_in_block = int(np.count_nonzero(covered[local_start:local_end])) 

687 if covered_in_block == 0: 

688 close_pending() 

689 continue 

690 actual = target64[local_start:local_end].reshape(-1) 

691 estimate = predicted_frames[local_start:local_end].reshape(-1) 

692 energy = float(np.sum(actual * actual, dtype=np.float64)) 

693 residual_vector = actual - estimate 

694 residual = float(np.sum(residual_vector * residual_vector, 

695 dtype=np.float64)) 

696 normalized_rmse = (math.sqrt(residual / energy) if energy >= 1e-12 

697 else (0.0 if residual < 1e-12 else float("inf"))) 

698 block_rms = math.sqrt(energy / max(actual.size, 1)) 

699 is_mismatch = ( 

700 normalized_rmse >= LOCAL_MISMATCH_RMSE and 

701 block_rms >= LOCAL_MISMATCH_MIN_RMS and 

702 covered_in_block >= max(1, (local_end - local_start) // 2)) 

703 if normalized_rmse <= 0.05: 

704 matched_frames += covered_in_block 

705 if is_mismatch: 

706 start_seconds = (block_start_frame + local_start) / rate 

707 end_seconds = (block_start_frame + local_end) / rate 

708 if (pending_region is not None and 

709 start_seconds <= pending_region["end_seconds"] + 1.5 / rate): 

710 pending_region["end_seconds"] = end_seconds 

711 pending_region["peak_normalized_rmse"] = max( 

712 pending_region["peak_normalized_rmse"], normalized_rmse) 

713 pending_region["block_count"] += 1 

714 else: 

715 close_pending() 

716 pending_region = { 

717 "start_seconds": start_seconds, 

718 "end_seconds": end_seconds, 

719 "peak_normalized_rmse": normalized_rmse, 

720 "block_count": 1, 

721 } 

722 else: 

723 close_pending() 

724 close_pending() 

725 

726 partition_metrics = _partition_result(fit_stat) 

727 validation_metrics = _partition_result(validation_stat) 

728 if partition_metrics is None or validation_metrics is None: 

729 return None 

730 total_frames = max(1, target_meta["total_frames"]) 

731 return { 

732 "declared_gains": declared_gains, 

733 "offsets_seconds": offsets, 

734 "analysis_sample_rate_hz": rate, 

735 "analysis_ranges_seconds": [[0.0, target_meta["duration_seconds"]]], 

736 "analysis_complete": True, 

737 "declaration_partition_normalized_rmse": partition_metrics[0], 

738 "declaration_partition_correlation": partition_metrics[1], 

739 "validation_normalized_rmse": validation_metrics[0], 

740 "validation_correlation": validation_metrics[1], 

741 "source_identifiability": _source_identifiability(gram), 

742 "max_source_correlation": _max_source_correlation(gram), 

743 "target_duration_seconds": target_meta["duration_seconds"], 

744 "covered_duration_seconds": covered_count / rate, 

745 "coverage_ratio": covered_count / total_frames, 

746 "matched_duration_seconds": matched_frames / rate, 

747 "matched_coverage_ratio": matched_frames / total_frames, 

748 "local_mismatch_regions": mismatch_regions, 

749 "longest_local_mismatch_seconds": max( 

750 (region["duration_seconds"] for region in mismatch_regions), default=0.0), 

751 }

754def _linear_status(assessment, match_rmse, match_correlation, 

755 contradiction_rmse=0.35, contradiction_correlation=0.5): 

756 if assessment["coverage_ratio"] < MIN_TARGET_COVERAGE: 

757 return "not_applicable", "insufficient_target_coverage" 

758 if assessment["local_mismatch_regions"]: 

759 return "contradicted", "sustained_local_mismatch" 

760 if assessment["max_source_correlation"] >= MAX_SOURCE_CORRELATION: 

761 return "not_applicable", "redundant_or_highly_correlated_sources" 

762 if assessment["source_identifiability"] < MIN_SOURCE_IDENTIFIABILITY: 

763 return "not_applicable", "source_contributions_not_identifiable" 

764 normalized_rmse = assessment["validation_normalized_rmse"] 

765 correlation = assessment["validation_correlation"] 

766 if normalized_rmse <= match_rmse and correlation >= match_correlation: 

767 return "matched", None 

768 if normalized_rmse >= contradiction_rmse or correlation < contradiction_correlation: 

769 return "contradicted", None 

770 return "not_applicable", "inconclusive_declared_reconstruction"

773def _segment_alternative_locations(audio, attrs): 

774 """Find full-file alternative positions with FFT coarse search.""" 

775 if attrs.get("source_start_seconds") is None or attrs.get("source_end_seconds") is None: 

776 return [] 

777 start = float(attrs["source_start_seconds"]) 

778 end = float(attrs["source_end_seconds"]) 

779 samples = _mono_resample(audio, SEGMENT_SCAN_RATE_HZ) 

780 window_start = audio.get("start_seconds", 0.0) 

781 first = round((start - window_start) * SEGMENT_SCAN_RATE_HZ) 

782 last = round((end - window_start) * SEGMENT_SCAN_RATE_HZ) 

783 if first < 0 or last > len(samples): 

784 return [] 

785 template = samples[first:last] 

786 if len(template) < 25 or len(samples) < len(template) + 25: 

787 return [] 

788 exclusion = max(1, round(0.1 * SEGMENT_SCAN_RATE_HZ)) 

789 alternatives = [] 

790 for score, candidate in _fft_normalized_candidates( 

791 template, samples, top_k=32, 

792 min_separation=max(exclusion, len(template) // 2)): 

793 if abs(candidate - first) <= exclusion: 

794 continue 

795 # Re-evaluate candidates in the time domain before reporting them. 

796 comparison = samples[candidate:candidate + len(template)] 

797 # Discard only the small FIR boundary region for the precise check. 

798 # Identical source material can otherwise differ at a cut because its 

799 # neighbouring samples (or file-edge padding) are different. 

800 trim = min(round(0.04 * SEGMENT_SCAN_RATE_HZ), 

801 max(0, (len(template) - 32) // 2)) 

802 precise_template = template[trim:len(template) - trim or None] 

803 precise_comparison = comparison[trim:len(comparison) - trim or None] 

804 if _correlation(precise_template, precise_comparison) >= \ 

805 SEGMENT_AMBIGUITY_CORRELATION: 

806 alternatives.append(round(window_start + candidate / SEGMENT_SCAN_RATE_HZ, 3)) 

807 if len(alternatives) >= 5: 

808 break 

809 return alternatives

812def _linear_pass(native, chain, *, relationship, target_type, scopes, rule,

814 check_segment_ambiguity=False): 

815 results = [] 

816 for target in native["artefacts"]: 

817 edges = [edge for edge in target.get("evidence", []) 

818 if edge["relationship_type"] == relationship] 

819 if not edges: 

820 continue 

821 base = _base(rule, pass_name, target, edges) 

822 errors = _declaration_errors(chain, target["artefact_hash"], edges) 

823 if errors: 

824 results.append(_input_error_result(base, errors)) 

825 continue 

826 if target["artefact_type"] != target_type: 

827 results.append({**base, "status": "not_applicable", 

828 "reason": f"Target is not declared as {target_type}."}) 

829 continue 

830 missing = _missing_parameters( 

831 edges, ("derivation_scope",) + LINEAR_DECLARATION_FIELDS)

834 **base, 

835 "status": "unavailable", 

836 "claim_status": "declared_unverified",

842 continue

844 for edge in edges}

872 "source_start_seconds"),

876 results.append({**base, "status": "unavailable", 

877 "claim_status": "declared_unverified", 

878 "reason": f"PCM analysis unavailable: {exc}", 

879 **({"reason_code": exc.reason_code, 

880 "analysis_complete": False} 

881 if isinstance(exc, ResourceBudgetExceeded) else {}), 

882 "content_match_is_provenance": False}) 

883 continue 

884 if assessed is None: 

885 results.append({**base, "status": "unavailable", 

886 "claim_status": "declared_unverified", 

887 "reason": "There is insufficient non-silent audio for a stable comparison.", 

888 "content_match_is_provenance": False}) 

889 continue

894 reason = "Held-out target blocks are reconstructed by the declared aligned linear sources within tolerance."

896 reason = "A sustained, non-silent target interval contradicts the declared aligned constant-gain model." 

897 elif status == "contradicted": 

898 reason = "Held-out target blocks contradict the explicitly declared aligned constant-gain model." 

899 elif review_reason == "insufficient_target_coverage": 

900 reason = "The declared sources do not cover enough of the target for a reliable match." 

901 elif review_reason == "redundant_or_highly_correlated_sources":

904 reason = "The submitted sources cannot be distinguished well enough to verify their declared individual contributions." 

905 elif review_reason == "ambiguous_source_segment_location": 

906 reason = "The declared source segment has near-identical alternatives at other positions." 

907 else: 

908 reason = "The independent holdout fit is inconclusive; nonlinear or undocumented processing may be present." 

909 results.append({**base, "status": status, 

910 "claim_status": _claim_status(status), "reason": reason, 

911 "verified_scope": next(iter(declared_scopes)), 

912 "declared_gains": [round(value, 6) 

913 for value in assessed["declared_gains"]], 

914 "parameter_source": "submitter_declared", 

915 "offsets_seconds": assessed["offsets_seconds"], 

916 "normalized_rmse": round(assessed["validation_normalized_rmse"], 6), 

917 "correlation": round(assessed["validation_correlation"], 6), 

918 "declaration_partition_normalized_rmse": round( 

919 assessed["declaration_partition_normalized_rmse"], 6), 

920 "declaration_partition_correlation": round( 

921 assessed["declaration_partition_correlation"], 6), 

922 "validation_normalized_rmse": round( 

923 assessed["validation_normalized_rmse"], 6), 

924 "validation_correlation": round(assessed["validation_correlation"], 6), 

925 "source_identifiability": round( 

926 assessed["source_identifiability"], 8), 

927 "gain_estimation_performed": False, 

928 "max_source_correlation": round( 

929 assessed["max_source_correlation"], 6), 

930 "target_duration_seconds": round( 

931 assessed["target_duration_seconds"], 6), 

932 "covered_duration_seconds": round( 

933 assessed["covered_duration_seconds"], 6), 

934 "coverage_ratio": round(assessed["coverage_ratio"], 6), 

935 "matched_duration_seconds": round( 

936 assessed["matched_duration_seconds"], 6), 

937 "matched_coverage_ratio": round( 

938 assessed["matched_coverage_ratio"], 6), 

939 "local_mismatch_regions": [{ 

940 "start_seconds": round(region["start_seconds"], 6), 

941 "end_seconds": round(region["end_seconds"], 6), 

942 "duration_seconds": round(region["duration_seconds"], 6), 

943 "peak_normalized_rmse": round( 

944 region["peak_normalized_rmse"], 6),

954 "analysis_ranges_seconds"],

1186def _moving_average(samples, window): 

1187 """Centred, boundary-shrinking moving average without sample loops.""" 

1188 values = np.asarray(samples, dtype=np.float32) 

1189 count = values.size 

1190 if window <= 1 or count == 0: 

1191 return values.copy() 

1192 half = window // 2 

1193 positions = np.arange(count) 

1194 low = np.maximum(0, positions - half) 

1195 high = np.minimum(count, positions + (window - half)) 

1196 prefix = np.concatenate((np.zeros(1, dtype=np.float64), 

1197 np.cumsum(values, dtype=np.float64))) 

1198 return ((prefix[high] - prefix[low]) / (high - low)).astype(np.float32)

1247def _correlation(left, right): 

1248 count = min(len(left), len(right)) 

1249 if count < 32: 

1250 return 0.0 

1251 left_values = np.asarray(left[:count], dtype=np.float64) 

1252 right_values = np.asarray(right[:count], dtype=np.float64) 

1253 left_values = left_values - np.mean(left_values) 

1254 right_values = right_values - np.mean(right_values) 

1255 denominator = float(np.linalg.norm(left_values) * np.linalg.norm(right_values)) 

1256 numerator = float(np.sum(left_values * right_values, dtype=np.float64)) 

1257 return numerator / denominator if denominator > 1e-12 else 0.0

1260def _fft_normalized_candidates(query, haystack, top_k=8, min_separation=1): 

1261 """Return top linear normalized cross-correlation locations via FFT.""" 

1262 query = np.asarray(query, dtype=np.float64).reshape(-1) 

1263 haystack = np.asarray(haystack, dtype=np.float64).reshape(-1) 

1264 width = query.size 

1265 candidate_count = haystack.size - width + 1 

1266 if width < 8 or candidate_count < 1: 

1267 return [] 

1268 centred = query - np.mean(query) 

1269 query_energy = float(np.sum(centred * centred, dtype=np.float64)) 

1270 if query_energy < 1e-12: 

1271 return [] 

1272 convolution_size = haystack.size + width - 1 

1273 fft_size = 1 << (convolution_size - 1).bit_length() 

1274 convolution = np.fft.irfft( 

1275 np.fft.rfft(haystack, fft_size) * 

1276 np.fft.rfft(centred[::-1], fft_size), fft_size) 

1277 numerator = convolution[width - 1:width - 1 + candidate_count] 

1278 prefix = np.concatenate(([0.0], np.cumsum(haystack, dtype=np.float64))) 

1279 square_prefix = np.concatenate( 

1280 ([0.0], np.cumsum(haystack * haystack, dtype=np.float64))) 

1281 local_sum = prefix[width:] - prefix[:-width] 

1282 local_square = square_prefix[width:] - square_prefix[:-width] 

1283 local_variance = np.maximum(local_square - local_sum * local_sum / width, 0.0) 

1284 denominator = np.sqrt(query_energy * local_variance) 

1285 scores = np.divide(numerator, denominator, out=np.zeros_like(numerator), 

1286 where=denominator > 1e-12) 

1287 ranking = np.abs(scores).copy() 

1288 candidates = [] 

1289 separation = max(1, int(min_separation)) 

1290 for _ in range(min(top_k, candidate_count)): 

1291 index = int(np.argmax(ranking)) 

1292 if not math.isfinite(ranking[index]) or ranking[index] < 0: 

1293 break 

1294 candidates.append((float(scores[index]), index)) 

1295 ranking[max(0, index - separation): 

1296 min(candidate_count, index + separation + 1)] = -1.0 

1297 return candidates

1366def _block_rms(samples, block_size): 

1367 values = np.asarray(samples, dtype=np.float32).reshape(-1) 

1368 if values.size == 0: 

1369 return np.empty(0, dtype=np.float32) 

1370 full = values.size // block_size 

1371 rows = values[:full * block_size].reshape(full, block_size) 

1372 result = np.sqrt(np.mean(rows * rows, axis=1)) if full else np.empty(0) 

1373 remainder = values[full * block_size:] 

1374 if remainder.size >= block_size // 2: 

1375 result = np.concatenate((result, [np.sqrt(np.mean(remainder * remainder))])) 

1376 return np.asarray(result, dtype=np.float32)
