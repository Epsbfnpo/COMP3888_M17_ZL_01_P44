26def _aaf_takes(response): 

27 """Source seconds each AAF-used recorder take contributes, keyed by the take's hash. 

28 

29 AAF ranges are tape timecode; subtracting the take's recorder start gives 

30 seconds into the file. Duplicate ranges (one per channel slot) are merged. 

31 A range more than one frame outside the recording is left to the core 

32 aaf_source_range_within_recorded_timecode check and not analysed. 

33 """ 

34 recorders = {} 

35 for row in response.get("findings", []): 

36 recorder = row.get("recorder_metadata") or {} 

37 if (row.get("code") in {"WAV_OBSERVATION", "AUDIO_OBSERVATION"} and recorder.get("present") 

38 and recorder.get("file_name")): 

39 recorders.setdefault((_norm(recorder.get("tape")), _norm(recorder["file_name"])), []).append( 

40 (row["evidence_hash"], recorder)) 

41 ranges, durations = {}, {} 

42 for row in response.get("findings", []): 

43 if not (row.get("code") == "PARSER_OBSERVATION" and row.get("parser") == "aaf" 

44 and row.get("parser_status") in {"parsed", "partial"} and row.get("parser_result")): 

45 continue 

46 for reference in row["parser_result"].get("references", []): 

47 matches = recorders.get((_norm(reference.get("tape")), _norm(reference.get("file_name"))), []) 

48 if len(matches) != 1: 48 ↛ 49line 48 didn't jump to line 49 because the condition on line 48 was never true

49 continue # unsubmitted or ambiguous take: the core reports it 

50 digest, recorder = matches[0] 

51 start, duration = recorder.get("start_timecode_seconds"), recorder.get("duration_seconds") 

52 if start is None or not duration: 52 ↛ 53line 52 didn't jump to line 53 because the condition on line 52 was never true

53 continue 

54 for item in reference.get("source_ranges", []): 

55 frame = 1 / item["edit_rate"] 

56 begin, end = item["start_seconds"] - start, item["end_seconds"] - start 

57 if begin < -frame or end > duration + frame or end - begin <= 0: 

58 continue 

59 key = (round(max(begin, 0.0), 6), round(min(end, duration), 6)) 

60 ranges.setdefault(digest, {}).setdefault(key, reference.get("master_mob")) 

61 durations[digest] = float(duration) 

62 return {digest: {"duration_seconds": durations[digest], "ranges": sorted(items.items())} 

63 for digest, items in ranges.items()}

157DERIVED = "derived_from_declared_source_duration" 

158AAF_UNUSED = "aaf_unused_range" 

159# Editors keep handles around each used clip; seconds this close to a used 

160# range are not counted as unused. 

161AAF_HANDLE_SECONDS = 0.5 

162# Shorter unused pieces carry too few landmarks to fingerprint reliably. 

163MINIMUM_UNUSED_SECONDS = 3.0 

164RANGE_SEARCH = "declared_range_search" 

165# Fingerprint providers that can locate a target window anywhere in the source. 

166SEARCH_PROVIDERS = ("audfprint", "spectromap")

169def _unused_pieces(duration, used): 

170 pieces, cursor = [], 0.0 

171 for begin, end in sorted(used) + [(duration + AAF_HANDLE_SECONDS, duration + AAF_HANDLE_SECONDS)]: 

172 stop = min(begin - AAF_HANDLE_SECONDS, duration) 

173 if stop - cursor >= MINIMUM_UNUSED_SECONDS: 

174 pieces.append((round(cursor, 6), round(stop, 6))) 

175 cursor = max(cursor, end + AAF_HANDLE_SECONDS) 

176 return pieces

203def range_search_tasks(tasks): 

204 """Whole-source companions for edges that declare both source and target seconds. 

205 

206 The declared task compares only the declared source window. Its companion 

207 searches the whole source for the declared target window, so a false 

208 interval can be told apart from audio that is simply hard to fingerprint. 

209 It reads only the two files the edge already declares. 

210 """ 

211 out = [] 

212 for task in tasks: 

213 attrs = task["attributes"] 

214 if (task.get("unavailable_reason") or task.get("range_source") not in ("declared", DERIVED) 

215 or not all(type(attrs.get(key)) in (int, float) for key in RANGE_FIELDS)): 

216 continue 

217 out.append({**task, "task_id": task["task_id"] + ":search", "range_source": RANGE_SEARCH, 

218 "attributes": {"source_start_seconds": None, "source_end_seconds": None, 

219 "target_start_seconds": attrs["target_start_seconds"], 

220 "target_end_seconds": attrs["target_end_seconds"]}}) 

221 return out

224def _range_source(attrs): 

225 values = [attrs.get(key) for key in RANGE_FIELDS] 

226 if all(value is None for value in values): 

227 return "whole_file" 

228 numbers = all(type(value) in (int, float) for value in values[:3])
