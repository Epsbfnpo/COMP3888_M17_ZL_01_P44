25def _open(path): 

26 """(descriptor, frame reader) for a WAV, or a FLAC/MP3 decoded by soundfile.""" 

27 container = detect(path).container 

28 if container not in ("FLAC", "MP3"): 

29 descriptor = inspect_wav(Path(path)) 

30 return descriptor, lambda first, last: (block for _, block in read_frames(path, descriptor, first, last)) 

31 import io 

32 import soundfile 

33 source, tolerance = str(path), 0 

34 try: 

35 if container == "MP3": 

36 # libsndfile estimates the length of an MP3 without a Xing header 

37 # from the bytes it is given and stops decoding there, so a large 

38 # ID3 tag (a C2PA manifest) skews it. Only the MPEG frames are 

39 # decoded, and the expected length is the gapless one the file 

40 # states, otherwise every frame's samples, as the frame walk measures. 

41 mp3 = inspect_mp3(path) 

42 with open(path, "rb") as handle: 

43 handle.seek(mp3.audio_offset) 

44 frames_only = handle.read(mp3.audio_end - mp3.audio_offset) 

45 source = None 

46 expected = mp3.total_samples or mp3.audio_frames * mp3.samples_per_frame 

47 if mp3.total_samples is None: 

48 # Without gapless information decoders differ by their start-up 

49 # delay (mpg123 drops 529 samples; ffmpeg does not): allow up to 

50 # two frames, far less than any real truncation. 

51 tolerance = 2 * mp3.samples_per_frame 

52 with soundfile.SoundFile(source if source else io.BytesIO(frames_only)) as audio: 

53 frames = audio.frames if container == "FLAC" else expected 

54 descriptor = DecodedAudio(audio.samplerate, audio.channels, frames, frames / audio.samplerate) 

55 except (RuntimeError, soundfile.LibsndfileError, Mp3Error) as exc: 

56 raise AnalysisUnavailable("decoder_unavailable", f"The {container} file could not be decoded: {exc}") from None

58 def blocks(first, last): 

59 size = max(1, descriptor.sample_rate * DECODE_BLOCK_SECONDS) 

60 with soundfile.SoundFile(source if source else io.BytesIO(frames_only)) as audio: 

61 audio.seek(first) 

62 remaining = last - first 

63 while remaining > 0: 

64 block = audio.read(min(size, remaining), dtype="float32", always_2d=True) 

65 if not len(block): 

66 break 

67 remaining -= len(block) 

68 yield np.ascontiguousarray(block) 

69 if remaining > tolerance: 

70 # For example a variable-bitrate MP3 without a Xing header, whose 

71 # length libsndfile guesses from the first frame: never analyse a 

72 # silently truncated signal. 

73 raise AnalysisUnavailable("decoder_length_mismatch", 

74 f"The decoder delivered {last - first - remaining} of {last - first} expected " 

75 f"samples of this {container} file; the audio was not analysed.") 

76 return descriptor, blocks

79def _decode_mono(path, start, end): 

80 descriptor, blocks = _open(path) 

81 start = 0.0 if start is None else start 

82 end = descriptor.duration_seconds if end is None else end 

83 checks = content_range_checks(start, end, descriptor.duration_seconds, descriptor.sample_rate) 

84 if any(not check["passed"] for check in checks): 

85 raise AnalysisUnavailable("invalid_analysis_range", "Declared time range is outside the submitted audio.") 

86 end = min(end, descriptor.duration_seconds) 

87 first = round(start * descriptor.sample_rate) 

88 last = min(descriptor.frame_count, round(end * descriptor.sample_rate)) 

89 chunks = list(blocks(first, last)) 

90 if not chunks: 90 ↛ 91line 90 didn't jump to line 91 because the condition on line 90 was never true

91 raise AnalysisUnavailable("insufficient_signal", "The selected audio range is empty.") 

92 frames = np.concatenate(chunks, axis=0) 

93 mono = np.mean(frames, axis=1) 

94 rms_channels = float(np.sqrt(np.mean(frames.astype(np.float64) ** 2))) 

95 rms_mono = float(np.sqrt(np.mean(mono.astype(np.float64) ** 2))) 

96 if rms_channels < 1e-9: 

97 raise AnalysisUnavailable("silent_signal", "Selected audio is silent; no content correspondence can be assessed.") 

98 if rms_mono < rms_channels * 1e-5: 98 ↛ 99line 98 didn't jump to line 99 because the condition on line 98 was never true

99 raise AnalysisUnavailable("channel_cancellation", "Mono conversion cancels the submitted channels; no non-match is inferred.") 

100 mono.setflags(write=False) 

101 return mono, descriptor, [float(start), float(end)]

104def _cached(cache, key, compute): 

105 # One worker runs every provider for a task, so each file range is decoded 

106 # once and each analysis rate is resampled once. Failures are cached too, 

107 # so every provider reports the same input limitation. 

108 if cache is None: 

109 return compute() 

110 if key not in cache: 

111 try: 

112 cache[key] = (True, compute()) 

113 except AnalysisUnavailable as exc: 

114 cache[key] = (False, exc) 

115 ok, value = cache[key] 

116 if not ok: 

117 raise value 

118 return value

121def read_audio(path, start, end, output_rate, cache=None): 

122 from scipy.signal import resample_poly 

123 def resampled(): 

124 mono, descriptor, span = _cached(cache, ("mono", str(path), start, end), 

125 lambda: _decode_mono(path, start, end)) 

126 divisor = math.gcd(output_rate, descriptor.sample_rate) 

127 if output_rate != descriptor.sample_rate: 

128 mono = resample_poly(mono, output_rate // divisor, descriptor.sample_rate // divisor) 

129 samples = np.array(mono, dtype=np.float32) 

130 samples.setflags(write=False) # shared by providers through the cache 

131 return {"samples": samples, "sample_rate_hz": output_rate, 

132 "original_sample_rate_hz": descriptor.sample_rate, "channels": descriptor.channels, 

133 "range": span, "channel_policy": "arithmetic_mean"} 

134 return _cached(cache, ("rate", str(path), start, end, output_rate), resampled)

145 and values[3] is None and values[0] is not None and values[1] is not None) 

146 # A declared-range search reads the whole source against the declared target window. 

147 range_search = (task.get("range_source") == "declared_range_search" and values[0] is None 

148 and values[1] is None and values[2] is not None and values[3] is not None) 

149 if (not aaf_source_only and not range_search and any(value is not None for value in values)
