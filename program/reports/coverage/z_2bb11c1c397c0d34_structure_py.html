1"""Per-file multiscale acoustic-change segmentation, without musical names. 

2 

3Local checkerboard novelty is computed from pooled chroma/MFCC features. 

4Unlike a beat grid, boundary positions and the number of sections are inferred 

5from the audio. The mean-difference form below is the linear-kernel equivalent 

6of within-block minus cross-block self-similarity (Foote-style novelty).

14def _unit(features): 

15 return features / np.maximum(np.linalg.norm(features, axis=0, keepdims=True), 1e-12)

18def section_features(chroma, timbre, frame_seconds, span, aggregation_seconds=0.5): 

19 """Time pooling is independent of the beat tracker and comparison target.""" 

20 length = chroma.shape[1] 

21 stride = max(1, round(aggregation_seconds / frame_seconds)) 

22 starts = np.arange(0, length, stride) 

23 pool = lambda features: np.stack([np.median(features[:, first:first + stride], axis=1) 

24 for first in starts], axis=1) 

25 harmony, colour = pool(chroma), pool(timbre) 

26 centre = np.median(colour, axis=1, keepdims=True) 

27 scale = np.maximum(np.median(np.abs(colour - centre), axis=1, keepdims=True) * 1.4826, 1.0) 

28 colour = _unit(np.clip((colour - centre) / scale, -4, 4)) 

29 features = _unit(np.vstack((0.7 * _unit(harmony), 0.3 * colour))) 

30 seconds = np.minimum(starts * frame_seconds + span[0], span[1]) 

31 return features, seconds

34def novelty_curve(features, seconds, context_seconds=4.0): 

35 """Multiscale local self-similarity contrast, without a forced section count.""" 

36 count = features.shape[1]

38 return np.zeros(count) 

39 step = max(float(np.median(np.diff(seconds))), 1e-6) 

40 curves = [] 

41 cumulative = np.column_stack((np.zeros(features.shape[0]), np.cumsum(features, axis=1))) 

42 for scale in (0.5, 1.0, 2.0): 

43 width = max(2, round(context_seconds * scale / step)) 

44 curve = np.zeros(count) 

45 for index in range(1, count): 

46 left, right = max(0, index - width), min(count, index + width) 

47 if min(index - left, right - index) < width / 2: 

48 continue 

49 before = (cumulative[:, index] - cumulative[:, left]) / (index - left) 

50 after = (cumulative[:, right] - cumulative[:, index]) / (right - index) 

51 curve[index] = float(np.sum((before - after) ** 2)) 

52 curves.append(curve) 

53 return gaussian_filter1d(np.mean(curves, axis=0), sigma=max(0.5, 0.35 / step), mode="nearest")
