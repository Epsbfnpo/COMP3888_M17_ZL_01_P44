1"""FLAC and MP3 observations in the same shape as the WAV ones. 

2 

3``observed`` holds the measured technical properties that submitter claims are 

4compared with, and ``embedded_metadata`` the metadata found in the file. Keys 

5that WAV consumers already read (``broadcast_extension``, ``info``, 

6``recorder_metadata``) keep their meaning, so the chronology, sound-report and 

7AAF checks work on FLAC takes unchanged. A property a format cannot carry is 

8``None``, and the comparisons that need it are reported as unavailable. 

9"""

16# The evaluator is designed and calibrated for WAV/BWF. FLAC and MP3 receive 

17# the checks implemented today; every other check family is reported as not 

18# assessed, never as a contradiction and never in a score. Each entry says, 

19# per format, whether the family is implemented ("implemented"), implemented 

20# only when the file carries the information ("if_in_file"), not implemented 

21# by this program ("not_implemented_for_format", a program limitation), or 

22# impossible for the format ("format_cannot_carry"). 

23FORMAT_SCOPE = { 

24 "hash_binding": {"WAV": "implemented", "FLAC": "implemented", "MP3": "implemented"}, 

25 "container_structure": {"WAV": "implemented", "FLAC": "implemented", "MP3": "implemented"}, 

26 "declared_technical_values": {"WAV": "implemented", "FLAC": "implemented", "MP3": "implemented"}, 

27 "declared_bit_depth": {"WAV": "implemented", "FLAC": "implemented", "MP3": "format_cannot_carry"}, 

28 "declared_duration": {"WAV": "implemented", "FLAC": "if_in_file", "MP3": "if_in_file"}, 

29 "decoded_sample_integrity": {"WAV": "implemented", "FLAC": "if_in_file", "MP3": "format_cannot_carry"}, 

30 "recorder_metadata": {"WAV": "implemented", "FLAC": "if_in_file", "MP3": "format_cannot_carry"}, 

31 "bwf_chronology": {"WAV": "implemented", "FLAC": "if_in_file", "MP3": "format_cannot_carry"}, 

32 "tag_dates": {"WAV": "not_applicable", "FLAC": "not_implemented_for_format", 

33 "MP3": "not_implemented_for_format"}, 

34 "embedded_c2pa": {"WAV": "implemented", "FLAC": "implemented", "MP3": "implemented"}, 

35 "lossless_content": {"WAV": "implemented", "FLAC": "implemented", "MP3": "format_cannot_carry"}, 

36} 

37# Which family each declarable technical field belongs to. 

38TECHNICAL_FAMILY = {"sample_rate_hz": "declared_technical_values", "channels": "declared_technical_values", 

39 "file_size_bytes": "declared_technical_values", "format": "declared_technical_values", 

40 "codec_subtype": "declared_technical_values", "bit_depth": "declared_bit_depth", 

41 "duration_seconds": "declared_duration"} 

42LIMITATION_OF = {"not_implemented_for_format": "program_scope", "format_cannot_carry": "format", 

43 "not_in_file": "file"} 

44_REASONS = { 

45 ("declared_bit_depth", "MP3"): "MP3 has no bit depth; declared bit depths are not compared.", 

46 ("declared_duration", "FLAC"): "STREAMINFO does not record the total number of samples, so the length " 

47 "is unknown and duration claims are not compared.", 

48 ("declared_duration", "MP3"): "The MP3 has no Xing/Info header with encoder delay and padding, so its " 

49 "exact length is unknown; duration comparisons are not made.", 

50 ("decoded_sample_integrity", "FLAC"): "The encoder stored no MD5 of the audio in STREAMINFO, so the decoded " 

51 "samples cannot be checked against it; every frame is still decoded " 

52 "and checked by audio_samples_valid.", 

53 ("decoded_sample_integrity", "MP3"): "MP3 stores no checksum of the audio, so decoded samples cannot be " 

54 "checked against the encoder's; every frame is still decoded and " 

55 "checked by audio_samples_valid.", 

56 ("recorder_metadata", "FLAC"): "This FLAC carries no WAV chunks (bext/iXML), so sound-report and AAF " 

57 "comparisons with recorder fields are not possible.", 

58 ("recorder_metadata", "MP3"): "MP3 has no standard field for recorder metadata (bext/iXML), so " 

59 "sound-report and AAF comparisons with recorder fields are not possible.", 

60 ("bwf_chronology", "FLAC"): "Without kept BWF chunks there is no origination date to compare.", 

61 ("bwf_chronology", "MP3"): "MP3 carries no BWF origination date to compare.", 

62 ("tag_dates", "FLAC"): "Dates in Vorbis comments are not read; this evaluator compares BWF dates only.", 

63 ("tag_dates", "MP3"): "Dates in ID3 tags (TDRC) are not read; this evaluator compares BWF dates only.", 

64 ("lossless_content", "MP3"): "MP3 is lossy: decoded samples differ from the original recording, so samples " 

65 "are never compared exactly and audio diagnostics compare an approximation.",

69def limited_checks(container, *, exact_duration=True, md5_stored=True, recorder_kept=True): 

70 """Check families not assessed on this FLAC or MP3 file, each as an unavailable row with its reason. 

71 

72 The keyword arguments say whether the file carries what an ``if_in_file`` 

73 family needs. Families implemented for the format are not listed. 

74 """ 

75 present = {"declared_duration": exact_duration, "decoded_sample_integrity": md5_stored, 

76 "recorder_metadata": recorder_kept, "bwf_chronology": recorder_kept} 

77 rows = [] 

78 for family, scope in FORMAT_SCOPE.items(): 

79 status = scope[container] 

80 if status == "if_in_file": 

81 if present.get(family, True): 

82 continue 

83 status = "not_in_file" 

84 if status in ("implemented", "not_applicable"): 

85 continue 

86 rows.append({"check_family": family, "status": "unavailable", "reason_code": status, 

87 "limitation": LIMITATION_OF[status], "reason": _REASONS[(family, container)]}) 

88 return rows

107def flac_embedded(descriptor): 

108 """Metadata summary plus the WAV chunks a FLAC kept from its source.""" 

109 chunks = dict(descriptor.foreign_chunks) 

110 bext = _parse_bext(chunks[b"bext"]) if b"bext" in chunks else None 

111 info = _parse_list_info(chunks[b"LIST"]) if b"LIST" in chunks else None 

112 return { 

113 "container": "FLAC", 

114 **descriptor.metadata, 

115 "broadcast_extension": dataclasses.asdict(bext) if bext else None, 

116 "info": info or {}, 

117 "id3v2_tags": list(descriptor.id3_tags), 

118 "c2pa_container_present": descriptor.c2pa_container_present, 

119 "warnings": list(descriptor.warnings), 

120 }

123def flac_recorder(descriptor, embedded): 

124 """Recorder metadata from kept bext/iXML chunks; absent when the encoder dropped them.""" 

125 chunks = dict(descriptor.foreign_chunks) 

126 if b"bext" not in chunks and b"iXML" not in chunks: 

127 return {"present": False, "bwf_kind": None, "warnings": [], 

128 "foreign_metadata_kept": False} 

129 duration = descriptor.duration_seconds 

130 result = recorder_fields_from_chunks(embedded.get("broadcast_extension"), chunks.get(b"iXML"), 

131 descriptor.sample_rate, duration) 

132 return {**result, "foreign_metadata_kept": True}
