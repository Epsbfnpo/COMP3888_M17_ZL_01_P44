1"""How hard the checked cross-evidence would be to forge; scales Integrity confidence only. 

2 

3A forge cost (1-5) says how much consistent work it would take to fake the 

4evidence behind a cross-evidence fact. It never changes the Integrity value: 

5the value says what the checks found, the confidence how strongly the checked 

6evidence supports it. Tiers keep their role as the release gate; forge cost 

7grades the evidence within and below them, and the profile enforces that no 

8submitter document outranks recorder metadata. 

9 

10Each subject (one recording, one document collection) takes the cost of its 

11strongest assessed fact, and the strength is the mean over subjects. A plain 

12mean over facts would let an extra, weaker corroborating document lower 

13confidence; per-subject maxima cannot. A fact whose documents or recording are 

14not hash-bound gets the unbound cost, whatever its tier. 

15""" 

16 

17from music_target.evidence_tiers import RECORDER_METADATA, SIGNED_ATTESTATION

19# Identity keys that name the check rather than the subject it is about. 

20_RULE_KEYS = ("rule", "claim") 

21 

22 

23def _subject_key(identity): 

24 """The artefacts a fact is about, whatever the check calls them. 

25 

26 A cue sheet names a take ``asset_hash`` and a sound report ``audio_hash``; 

27 both are facts about the same recording and must share its subject. 

28 """ 

29 hashes, other = set(), [] 

30 for name, value in identity.items(): 

31 if name in _RULE_KEYS: 

32 continue 

33 if name.endswith("_hash"): 

34 hashes.add(value) 

35 elif name.endswith("_hashes") and isinstance(value, list): 

36 hashes.update(value) 

37 else: 

38 other.append((name, str(value))) 

39 return tuple(sorted(hashes)), tuple(sorted(other))

42def fact_cost(fact, artefact_types, bound_hashes, config): 

43 """(cost, basis) for one assessed cross-evidence fact.""" 

44 identity = fact.get("identity") or {} 

45 documents = fact.get("document_hashes") or [] 

46 audio = identity.get("audio_hash") 

47 if any(digest not in bound_hashes for digest in documents) or (audio and audio not in bound_hashes): 

48 return config["unbound_source_cost"], "unbound_source" 

49 tiers = set(fact.get("assessed_evidence_tiers") or []) 

50 for tier in (SIGNED_ATTESTATION, RECORDER_METADATA): 

51 if tier in tiers: 

52 return config["tier_costs"][tier], tier 

53 costs = config["submitter_document_costs"] 

54 typed = [(costs[artefact_types[digest]], "submitter_document:" + artefact_types[digest]) 

55 for digest in documents if artefact_types.get(digest) in costs] 

56 if typed: 

57 return max(typed) 

58 return config["submitter_document_default"], "submitter_document"

61def evidence_strength(cross_facts, artefact_types, bound_hashes, config): 

62 """Mean over subjects of the strongest assessed fact's forge cost, and the confidence factor.""" 

63 subjects = {} 

64 for fact in cross_facts: 

65 if not isinstance(fact.get("passed"), bool): 

66 continue 

67 cost, basis = fact_cost(fact, artefact_types, bound_hashes, config) 

68 key = _subject_key(fact.get("identity") or {}) 

69 if key not in subjects or cost > subjects[key]["forge_cost"]: 

70 subjects[key] = {"subject": {name: value for name, value in (fact.get("identity") or {}).items() 

71 if name not in _RULE_KEYS}, 

72 "forge_cost": cost, "basis": basis, "rule": fact.get("rule")} 

73 scale, floor = config["scale_max"], config["strength_floor"] 

74 if subjects: 

75 mean = sum(item["forge_cost"] for item in subjects.values()) / len(subjects) 

76 basis = "assessed_cross_evidence" 

77 else: 

78 # Only the submission's own claims were checked: submitter-level evidence. 

79 mean, basis = config["submitter_document_default"], "no_cross_evidence" 

80 by_basis = {} 

81 for item in subjects.values(): 

82 by_basis[item["basis"]] = by_basis.get(item["basis"], 0) + 1 

83 return {"applies_to": "integrity_confidence_only", "basis": basis, 

84 "subject_count": len(subjects), "mean_subject_forge_cost": round(mean, 3), 

85 "scale_max": scale, "strength_floor": floor, 

86 "confidence_factor": round(floor + (1 - floor) * mean / scale, 3), 

87 "subjects_by_basis": dict(sorted(by_basis.items())), 

88 "subjects": sorted(subjects.values(), key=lambda item: (-item["forge_cost"], str(item["subject"]))), 

89 "derivation": config.get("derivation")}

92def describe(strength, layers, denominator): 

93 """One sentence for the Integrity reasoning.""" 

94 if strength["basis"] == "no_cross_evidence": 

95 source = ("no cross-evidence was assessed, so it rests on the submission's own claims " 

96 f"(forge cost {strength['mean_subject_forge_cost']:g} of {strength['scale_max']})") 

97 else: 

98 parts = ", ".join(f"{count} {name.replace('_', ' ')}" 

99 for name, count in strength["subjects_by_basis"].items()) 

100 source = (f"the strongest checked evidence per subject has an average forge cost of " 

101 f"{strength['mean_subject_forge_cost']:.2f} of {strength['scale_max']} ({parts})") 

102 return (f"Integrity confidence reflects how many of the three evidence layers were checked " 

103 f"({layers}/{denominator}) and how hard the checked evidence would be to forge: {source}, " 

104 f"a factor of {strength['confidence_factor']:.2f}. Forge cost changes confidence only, never the value. ")
