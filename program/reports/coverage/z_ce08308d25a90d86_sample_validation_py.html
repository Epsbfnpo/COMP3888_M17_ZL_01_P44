1"""Full-length sample check of every bound audio file in a normal evaluation. 

2 

3The structure checks (RIFF chunks, FLAC metadata, MP3 frame headers) do not 

4read the samples, so a float WAV holding NaN or Infinity, or audio a decoder 

5cannot get through, used to pass the file check. This module reads the whole 

6stream in bounded blocks and returns one of three outcomes: 

7 

8- ``valid``: every frame was read and every sample is finite; 

9- ``invalid``: a non-finite sample was found or the decoder failed part-way; 

10- ``not_completed``: the scan stopped (caller's time limit, file changed, no 

11 decoder). This is never a pass. 

12 

13WAV float samples are checked as stored (float32 or float64), before any 

14conversion, so a float64 value that only overflows float32 is not reported 

15as damage. Results are recorded with ``CHECK_VERSION`` and the decoder used. 

16"""

58def scan_wav(path, descriptor, deadline=None): 

59 """Read every PCM frame of an inspected WAV; check float samples as stored.""" 

60 path = Path(path) 

61 float_dtype = ({32: "<f4", 64: "<f8"}.get(descriptor.container_bits) 

62 if descriptor.format_kind == "float" else None) 

63 record = {"check_version": CHECK_VERSION, "decoder": "wav_io", 

64 "values_checked": f"stored_float{descriptor.container_bits}" if float_dtype else "integer_pcm", 

65 "frame_count": descriptor.frame_count, "frames_checked": 0, 

66 "non_finite_samples": 0, "first_non_finite": None} 

67 stat = path.stat() 

68 if (stat.st_size != descriptor.file_size 

69 or (descriptor.mtime_ns and stat.st_mtime_ns != descriptor.mtime_ns)): 

70 return _not_completed(record, "file_changed_after_inspection", "The file changed after it was inspected.") 

71 block_frames = max(1, MAX_DECODE_BLOCK_BYTES // descriptor.block_align) 

72 with path.open("rb") as handle: 

73 handle.seek(descriptor.data_offset) 

74 while record["frames_checked"] < descriptor.frame_count: 

75 if deadline is not None and time.time() >= deadline: 

76 return _finish(_not_completed( 

77 record, "caller_deadline", 

78 "The caller's time limit was reached before every sample was read.")) 

79 count = min(block_frames, descriptor.frame_count - record["frames_checked"]) 

80 raw = handle.read(count * descriptor.block_align) 

81 if len(raw) != count * descriptor.block_align: 81 ↛ 82line 81 didn't jump to line 82 because the condition on line 81 was never true

82 return _finish({**record, "status": "invalid", "reason_code": "truncated_pcm_frames", 

83 "reason": "The PCM data ended before the declared number of frames."}) 

84 if float_dtype:

86 _non_finite(block, record["frames_checked"], descriptor.sample_rate, record)

88 return _finish({**record, "status": "valid"})

92 """Decode a FLAC or MP3 with soundfile in blocks and check every sample is finite."""

121def from_flac_md5(md5): 

122 """Sample-check outcome of the FLAC MD5 pass, which already decodes every frame. 

123 

124 A decode failure is already a failed ``flac_md5_matches_decoded_audio``, 

125 so it returns None here rather than a second failure for the same fault. 

126 """ 

127 record = {"check_version": CHECK_VERSION, "decoder": "soundfile (FLAC MD5 pass)", 

128 "values_checked": "decoded_int32", "frame_count": md5.get("declared_total_samples"), 

129 "frames_checked": md5.get("decoded_frames") or 0, "non_finite_samples": 0, 

130 "first_non_finite": None} 

131 if md5.get("status") in ("matched", "mismatched"): 

132 # Integer FLAC samples are always finite once decoded. 

133 return {**record, "status": "valid"} 

134 if md5.get("status") == "failed": 

135 return None 

136 return _not_completed(record, md5.get("reason_code") or "not_decoded", 

137 md5.get("reason") or "The audio was not decoded.")

140def sample_check(digest, container, result): 

141 """The ``audio_samples_valid`` check row for one file.""" 

142 check = {"rule": RULE, "evidence_hash": digest, "source_pass": "WavMetadataPass", "format": container, 

143 **{key: value for key, value in result.items() if key != "status"}} 

144 if result["status"] == "valid": 

145 return {**check, "passed": True} 

146 if result["status"] == "invalid": 

147 return {**check, "passed": False} 

148 return {**check, "passed": None, "status": "not_completed"}
