312 reason and there is simply no manifest. A manifest location the SDK finds no

315 metadata = (wav_result or {}).get("embedded_metadata") 

316 if not c2pa.get("sdk_available") or not isinstance(metadata, dict): 

317 return c2pa # container unknown (parser failed): keep the SDK result 

318 if "c2pa_container_present" in metadata: 

319 has_chunk = bool(metadata["c2pa_container_present"]) 

320 else: 

321 has_chunk = any(item.get("chunk_id") == "C2PA" for item in metadata.get("other_chunks") or [])
