1"""Exact sample identity between lossless audio files. 

2 

3Two files hold the same audio when their decoded integer samples are equal at 

4the same sample rate, channel count and bit depth. The digest is SHA-256 over 

5the samples as FLAC defines them for its MD5: signed little-endian integers of 

6ceil(bits/8) bytes, channels interleaved (8-bit WAV is unsigned and is centred 

7first; EXTENSIBLE samples are left-justified in their container and are shifted 

8back to their valid bits). Float WAV has no integer representation and is not 

9compared. Identical samples describe an exact content relation; they do not 

10show who made either file or which came first. 

11"""

37def wav_sample_digest(path, descriptor, deadline=None, block_frames=65536): 

38 """Canonical sample SHA-256 of an integer-PCM WAV; ``deadline`` is a ``time.time()`` value.""" 

39 if descriptor.format_kind != "int": 

40 return _unavailable("float_samples", "Float samples have no integer representation to compare.")

42 width = (bits + 7) // 8 

43 remaining = descriptor.frame_count * descriptor.block_align 

44 digest = hashlib.sha256() 

45 with open(path, "rb") as handle: 

46 handle.seek(descriptor.data_offset) 

47 while remaining: 

48 if deadline is not None and time.time() >= deadline: 48 ↛ 49line 48 didn't jump to line 49 because the condition on line 48 was never true

49 return _unavailable("caller_deadline", 

50 "The caller's time limit was reached before every sample was read.") 

51 raw = handle.read(min(remaining, block_frames * descriptor.block_align)) 

52 if not raw: 52 ↛ 53line 52 didn't jump to line 53 because the condition on line 52 was never true

53 return _unavailable("truncated_data", "The data chunk ended before its declared size.") 

54 remaining -= len(raw) 

55 if container == bits and bits > 8: 

56 digest.update(raw) # already the canonical bytes 

57 continue 

58 values = _container_ints(raw, container) >> (container - bits) 

59 digest.update(values.astype("<i4").view(np.uint8).reshape(-1, 4)[:, :width].tobytes()) 

60 return {"status": "computed", "sha256": digest.hexdigest(), "frames": descriptor.frame_count, "bits": bits}
